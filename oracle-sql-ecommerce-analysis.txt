--Bu hissədə biz cədvələri yaradib içərisinə dəyərləri əlavə etdik
CREATE TABLE customers (
    customer_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    full_name   VARCHAR2(100),
    city        VARCHAR2(50),
    signup_date DATE
);

CREATE TABLE products (
    product_id   NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    product_name VARCHAR2(100),
    category     VARCHAR2(50),
    price        NUMBER(10,2)
);

CREATE TABLE orders (
    order_id    NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    customer_id NUMBER,
    order_date  DATE,
    CONSTRAINT fk_orders_customer
        FOREIGN KEY (customer_id)
        REFERENCES customers(customer_id)
);

CREATE TABLE order_items (
    item_id    NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    order_id   NUMBER,
    product_id NUMBER,
    quantity   NUMBER,
    CONSTRAINT fk_items_order
        FOREIGN KEY (order_id) REFERENCES orders(order_id),
    CONSTRAINT fk_items_product
        FOREIGN KEY (product_id) REFERENCES products(product_id)
);

--Müştərinin ümumi xərci + kateqoriya

SELECT
    c.full_name,
    SUM(oi.quantity * p.price) AS total_spent,
    CASE
        WHEN SUM(oi.quantity * p.price) > 2000 THEN 'VIP'
        WHEN SUM(oi.quantity * p.price) BETWEEN 1000 AND 2000 THEN 'Regular'
        ELSE 'New'
    END AS customer_type
FROM customers c
JOIN orders o ON c.customer_id = o.customer_id
JOIN order_items oi ON o.order_id = oi.order_id
JOIN products p ON oi.product_id = p.product_id
GROUP BY c.full_name;

--Top müştərilər (Analytic Function)

SELECT
    c.full_name,
    SUM(oi.quantity * p.price) AS total_spent,
    RANK() OVER (ORDER BY SUM(oi.quantity * p.price) DESC) AS rank_no
FROM customers c
JOIN orders o ON c.customer_id = o.customer_id
JOIN order_items oi ON o.order_id = oi.order_id
JOIN products p ON oi.product_id = p.product_id
GROUP BY c.full_name;

--Kateqoriya üzrə satış

SELECT
    p.category,
    SUM(oi.quantity * p.price) AS total_sales
FROM products p
JOIN order_items oi ON p.product_id = oi.product_id
GROUP BY p.category;

--ÜMUMİ SATIŞ (Total Revenue)
SELECT SUM(oi.quantity * p.price) AS total_revenue
FROM order_items oi
JOIN products p ON oi.product_id = p.product_id;

--ƏN ÇOX GƏLİR GƏTİRƏN KATEQORİYA
SELECT
    p.category,
    SUM(oi.quantity * p.price) AS total_sales
FROM products p
JOIN order_items oi ON p.product_id = oi.product_id
GROUP BY p.category
ORDER BY total_sales DESC;

--VIP MÜŞTƏRİ SAYI
SELECT COUNT(*) AS vip_count
FROM (
    SELECT
        c.customer_id,
        SUM(oi.quantity * p.price) AS total_spent
    FROM customers c
    JOIN orders o ON c.customer_id = o.customer_id
    JOIN order_items oi ON o.order_id = oi.order_id
    JOIN products p ON oi.product_id = p.product_id
    GROUP BY c.customer_id
    HAVING SUM(oi.quantity * p.price) > 2000
);

--Top 1 Müştərinin Ümumi Satışdakı Payı
WITH customer_spending AS (
    SELECT
        c.customer_id,
        SUM(oi.quantity * p.price) AS total_spent
    FROM customers c
    JOIN orders o ON c.customer_id = o.customer_id
    JOIN order_items oi ON o.order_id = oi.order_id
    JOIN products p ON oi.product_id = p.product_id
    GROUP BY c.customer_id
),
total_revenue AS (
    SELECT SUM(total_spent) AS grand_total
    FROM customer_spending
)
SELECT
    MAX(total_spent) AS top_customer_spent,
    ROUND(MAX(total_spent) / grand_total * 100, 2) AS percentage_share
FROM customer_spending, total_revenue;

--Monthly Sales Trend
--2024 Mart ayı ən yüksək satış olan aydır.
--Satışlar yanvar → fevral → mart üzrə artım trendi göstərir.
SELECT
    TO_CHAR(o.order_date, 'YYYY-MM') AS month,
    SUM(oi.quantity * p.price) AS monthly_revenue
FROM orders o
JOIN order_items oi ON o.order_id = oi.order_id
JOIN products p ON oi.product_id = p.product_id
GROUP BY TO_CHAR(o.order_date, 'YYYY-MM')
ORDER BY month;

--Average Order Value
--Average Order Value ≈ 1,131 AZN
--Orta sifariş məbləği yüksəkdir → premium məhsullar (Electronics) dominantdır.	
SELECT
    ROUND(SUM(oi.quantity * p.price) / COUNT(DISTINCT o.order_id), 2) AS avg_order_value
FROM orders o
JOIN order_items oi ON o.order_id = oi.order_id
JOIN products p ON oi.product_id = p.product_id;

--Top 3 Məhsul (Gəlirə görə)
--Laptop, iPhone və Samsung TV ən yüksək gəlir gətirən məhsullardır.
SELECT *
FROM (
    SELECT
        p.product_name,
        SUM(oi.quantity * p.price) AS total_sales,
        RANK() OVER (ORDER BY SUM(oi.quantity * p.price) DESC) AS rank_no
    FROM products p
    JOIN order_items oi ON p.product_id = oi.product_id
    GROUP BY p.product_name
)
WHERE rank_no <= 3;

--Electronics Revenue Contribution %
--Electronics ümumi satışın təxminən 70%+ hissəsini təşkil edir.
WITH category_sales AS (
    SELECT
        p.category,
        SUM(oi.quantity * p.price) AS total_sales
    FROM products p
    JOIN order_items oi ON p.product_id = oi.product_id
    GROUP BY p.category
),
total_sales AS (
    SELECT SUM(total_sales) AS grand_total
    FROM category_sales
)
SELECT
    category,
    ROUND(total_sales / grand_total * 100, 2) AS revenue_percentage
FROM category_sales, total_sales
ORDER BY revenue_percentage DESC;
